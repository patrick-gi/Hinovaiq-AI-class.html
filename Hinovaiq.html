// ---------- Profile and live-class schedule ---------- const LEARNING_SETTINGS_KEY = "hinovaiq-learning-settings-v1"; let profile = { name: "" }; let liveClasses = []; try { const saved = JSON.parse( localStorage.getItem(LEARNING_SETTINGS_KEY) || "null" ); if (saved && typeof saved === "object") { if (typeof saved.name === "string") { profile.name = saved.name.slice(0, 60); } if (Array.isArray(saved.classes)) { liveClasses = saved.classes.filter(session => session && typeof session.id === "string" && typeof session.title === "string" && typeof session.courseId === "string" && Number.isFinite(Date.parse(session.startsAt)) && Number.isFinite(session.duration) && session.duration >= 15 && session.duration <= 480 && validZoomURL(session.zoomURL) ); } } } catch { notify("Your profile or class schedule could not be loaded."); } if ($("#display-name")) { $("#display-name").value = profile.name; } function saveLearningSettings() { try { localStorage.setItem(LEARNING_SETTINGS_KEY, JSON.stringify({ name: profile.name, classes: liveClasses })); return true; } catch { notify("Changes could not be saved to this browser."); return false; } } // Accept HTTPS meeting URLs on zoom.us or its subdomains. // Reject unrelated domains and URLs containing embedded credentials. function validZoomURL(value) { try { const url = new URL(value); const host = url.hostname.toLowerCase(); const zoomHost = host === "zoom.us" || host.endsWith(".zoom.us"); const meetingPath = /^\/(?:j\/\d+|my\/[^/]+)\/?$/.test(url.pathname); if ( url.protocol !== "https:" || !zoomHost || !meetingPath || url.username || url.password || url.port ) { return null; } return url.href; } catch { return null; } } function classStatus(session) { const now = Date.now(); const start = Date.parse(session.startsAt); const end = start + session.duration * 60000; if (now >= end) return "Past"; if (now >= start) return "Scheduled now"; return "Upcoming"; } function formatClassTime(value) { return new Intl.DateTimeFormat(undefined, { weekday: "short", month: "short", day: "numeric", year: "numeric", hour: "numeric", minute: "2-digit", timeZoneName: "short" }).format(new Date(value)); } function liveClassCard(session) { const course = courses.find(c => c.id === session.courseId); const status = classStatus(session); const zoomURL = validZoomURL(session.zoomURL); return `
${status}

${escapeHTML(session.title)}

${escapeHTML(course?.title || "Learning session")}

${escapeHTML(formatClassTime(session.startsAt))}

${session.duration} minutes

${status !== "Past" && zoomURL ? ` Join on Zoom ↗ ` : ""}
`; } function renderLiveClasses() { const container = $("#live-class-list"); if (!container) return; const ordered = [...liveClasses].sort((a, b) => { const aPast = classStatus(a) === "Past"; const bPast = classStatus(b) === "Past"; // Upcoming/current classes first; most recent past classes last. if (aPast !== bPast) return aPast ? 1 : -1; return aPast ? Date.parse(b.startsAt) - Date.parse(a.startsAt) : Date.parse(a.startsAt) - Date.parse(b.startsAt); }); container.innerHTML = ordered.length ? ordered.map(liveClassCard).join("") : empty("No live classes scheduled yet."); } // Preserve the existing pages and add personalized Home content. function render() { renderWorkspace(); if (!signedIn || currentRoute().page !== "home") return; const welcomeTitle = $("#main .welcome h1"); if (welcomeTitle) { welcomeTitle.textContent = profile.name ? `Welcome back, ${profile.name}!` : "Welcome to HINOVAIQ!"; } const section = document.createElement("section"); section.id = "live-classes"; section.style.marginBottom = "1.5rem"; section.innerHTML = `

Live classes

Times use your device’s timezone. “Scheduled now” reflects the timetable, not Zoom meeting activity.

`; const welcome = $("#main .welcome"); if (welcome) welcome.after(section); else $("#main").prepend(section); renderLiveClasses(); } function openClassForm() { openModal("Schedule a live class", `

Enter the time in your device’s local timezone.

Create the meeting in Zoom, then paste its invitation link here. Include the passcode parameter if Zoom includes one.

`); $("#class-form").addEventListener("submit", event => { event.preventDefault(); const title = $("#class-title").value.trim(); const courseId = $("#class-course").value; const start = new Date($("#class-start").value); const duration = Number($("#class-duration").value); const zoomURL = validZoomURL($("#class-zoom").value.trim()); if (!title || !courses.some(course => course.id === courseId)) { notify("Enter a title and select a course."); return; } if (!Number.isFinite(start.getTime()) || start.getTime() <= Date.now()) { notify("Choose a future date and time."); return; } if (!Number.isInteger(duration) || duration < 15 || duration > 480) { notify("Duration must be a whole number between 15 and 480 minutes."); return; } if (!zoomURL) { notify("Enter a valid HTTPS Zoom meeting link on zoom.us."); return; } liveClasses.push({ id: uid(), title, courseId, startsAt: start.toISOString(), duration, zoomURL }); const saved = saveLearningSettings(); closeModal(); render(); if (saved) notify("Live class scheduled in this browser."); }); } document.addEventListener("click", event => { const button = event.target.closest("button[data-action]"); if (!button || !signedIn) return; const { action, id } = button.dataset; if (action === "schedule-class") { openClassForm(); } if (action === "remove-class") { openModal("Remove scheduled class?", `

This removes the class from this workspace. It does not cancel the meeting in Zoom.

`); } if (action === "confirm-remove-class") { liveClasses = liveClasses.filter(session => session.id !== id); const saved = saveLearningSettings(); closeModal(); render(); if (saved) notify("Class removed from this workspace."); } }); // Refresh timetable status without interrupting keyboard focus. setInterval(() => { const section = $("#live-classes"); if ( signedIn && currentRoute().page === "home" && !$("#modal").open && section && !section.contains(document.activeElement) ) { renderLiveClasses(); } }, 30000);
